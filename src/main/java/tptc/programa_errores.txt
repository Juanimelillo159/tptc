// Programa con errores léxicos intencionados

int main() {
    // Errores en números
    int numero1 = 123abc;     // Número seguido de letras
    double numero2 = 12.34.56; // Doble punto decimal
    
    // Errores en identificadores  
    int 123variable = 10;     // Identificador que empieza con número
    double var-iable = 5.0;   // Guión en identificador
    
    // Caracteres no reconocidos
    int resultado = 10 @ 5;   // @ no está en la gramática
    boolean test = numero1 # numero2; // # no está definido
    
    // Operadores mal formados
    if (numero1 === numero2) { // === no existe, solo ==
        int x = 5;
    }
    
    // Números mal formados
    double mal1 = .123;       // Empieza con punto
    double mal2 = 123.;       // Termina con punto
    
    return 0;
}
